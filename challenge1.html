Welke driehoek en welke boom wordt er zichtbaar als je deze scripts uitvoert m.b.v. de p5.js library?

let hoeken;
let x, y;
function setup() {
  createCanvas(800, 800);
  background(0);
  stroke(255);
  hoeken = [
    createVector(400, 50),
    createVector(50, 750),
    createVector(750, 750)
  ];
  x = random(width);
  y = random(height);
}
function draw() {
  for (let i = 0; i < 20; i++) {
    let h = random(hoeken);
    x = (x + h.x) / 2;
    y = (y + h.y) / 2;
    point(x, y);
  }
}
function setup() {
  createCanvas(800, 800);
  background(0);
  stroke(255);
  noFill();
  noLoop();
  translate(340, 750);
  recursiveFunction(120);
}
function wacht(ms) {
  return new Promise(klaar => setTimeout(klaar, ms));
}
async function recursiveFunction(s) {
  if (s < 5) return;
  rect(0, -s, s, s);
  await wacht(10);
  translate(0, -s);
  let k = s / sqrt(2);
  push();
  rotate(-PI / 4);
  await recursiveFunction(k);
  pop();
  push();
  translate(s / 2, -s / 2);
  rotate(PI / 4);
  await recursiveFunction(k);
  pop();
}
